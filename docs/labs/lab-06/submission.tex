\textbf{Detailed Finding Breakdown \& Technical Root Cause:}
\vspace{4pt}
\begin{enumerate}[leftmargin=*,itemsep=4pt,topsep=2pt,parsep=0pt]
  \item \textbf{\code{F-PB-01} (Floating-Point Decimal Validation Failure):}
  \begin{itemize}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
    \item \emph{Symptom:} In partner's test \code{UAT-PB-03}, submitting amount \code{1234567.89} returned 400 \code{"amount must have at most 2 decimal places"}.
    \item \emph{Root Cause:} Legacy validation used JavaScript floating-point arithmetic: \code{Math.round(val * 100) / 100 !== val}. Due to IEEE 754 precision errors, $1234567.89 \times 100 = 123456788.99999999$, producing false positive validation failures on large valid figures.
    \item \emph{Resolution:} Rewrote amount validation to perform string-based decimal splitting: \code{val.toString().split('.')[1]?.length <= 2}.
  \end{itemize}
  \item \textbf{\code{F-PB-02} (Oversized Body Unhandled Exception):}
  \begin{itemize}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
    \item \emph{Symptom:} In partner's exploratory stress test \code{UAT-PB-04}, sending an oversized 200 KB JSON body caused the server to return HTTP 500 \code{INTERNAL_ERROR} instead of a clean 4xx client rejection.
    \item \emph{Root Cause:} Express default \code{body-parser} threw error code \code{entity.too.large} (HTTP 413), which was unhandled by custom routing and fell through to the generic catch-all 500 internal error handler.
    \item \emph{Resolution:} Installed dedicated middleware intercepting \code{entity.too.large} and mapping it to a standard contract error: HTTP 400 with code \code{VALIDATION_ERROR}.
  \end{itemize}
\end{enumerate}

\partheader{Appendix A: Sprint 1 Test Case Detailed Specifications}{Specifications}

This appendix provides the full specifications of the four test cases authored for our Approval System (\code{UAT-OUM-01..04}), establishing the complete testing contract handed to the partner UAT team.

\subsection*{A.1 \code{UAT-OUM-01}: Normal Lifecycle Journey (Create $\to$ Review $\to$ Approve)}
\textbf{Purpose:} End-to-end verification of request lifecycle from creation by Requester to review and approval by assigned Reviewer.\\[1pt]
\textbf{Preconditions:} Deployed Cloud Run active; Requester (\code{requester1}) and Reviewer (\code{reviewer1}) accounts exist.\\[1pt]
\textbf{Execution Steps \& Checkpoints:}
\begin{enumerate}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
  \item \textbf{Login as Requester:} Submit credentials for \code{requester1} $\to$ Receive HTTP 200 with session cookie.
  \item \textbf{Submit Request:} \code{POST /api/requests} with Title \code{"Lab Equipment Q4"}, Amount \code{"123456.78"}, Category \code{"EQUIPMENT"}, Urgency \code{"STANDARD"}, and Justification text $\to$ Verify HTTP 201 Created, status \code{PENDING}, request number format \code{REQ-YYYYMMDD-XXXX}, and auto-assigned reviewer ID present.
  \item \textbf{Verify Reviewer Queue:} Login as \code{reviewer1} $\to$ \code{GET /api/requests/pending} $\to$ Newly created request is present in pending list.
  \item \textbf{Approve with Comment:} \code{POST /api/requests/:id/decision} with \code{{"decision":"APPROVE", "comment":"Budget approved within Q4 cap."}} $\to$ Verify HTTP 200, status changes to \code{APPROVED}, decision recorded in audit trail.
  \item \textbf{Requester Re-read:} Login as \code{requester1} $\to$ \code{GET /api/requests/:id} $\to$ Status is \code{APPROVED}, reviewer comment and timestamp visible.
\end{enumerate}

\subsection*{A.2 \code{UAT-OUM-02}: Rule Boundaries \& Validation Rejection}
\textbf{Purpose:} Validate that all invalid inputs and state boundary violations are strictly rejected with standard 400 errors.\\[1pt]
\textbf{Preconditions:} Authenticated as \code{requester1}.\\[1pt]
\textbf{Test Matrix \& Specific Expected Outcomes:}
\begin{itemize}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
  \item \textbf{Negative Amount:} \code{{"amount": "-500.00"}} $\to$ HTTP 400 \code{VALIDATION_ERROR} (\code{"amount must be positive"}).
  \item \textbf{Excessive Amount:} \code{{"amount": "10000000000.00"}} ($10$ Billion) $\to$ HTTP 400 \code{VALIDATION_ERROR} (exceeds $9,999,999,999.99$ limit).
  \item \textbf{Fractional Precision:} \code{{"amount": "100.999"}} $\to$ HTTP 400 \code{VALIDATION_ERROR} (\code{"amount must have at most 2 decimal places"}).
  \item \textbf{Title Length Limit:} Title with 151 characters $\to$ HTTP 400 \code{VALIDATION_ERROR} (\code{"title must not exceed 150 characters"}).
  \item \textbf{Empty Reject Comment:} Reviewer submits \code{{"decision":"REJECT", "comment":""}} $\to$ HTTP 400 \code{VALIDATION_ERROR} (\code{"comment required for rejection"}).
  \item \textbf{Double Decision on Finalized State:} Reviewer attempts second decision on already \code{APPROVED} request $\to$ HTTP 400 \code{INVALID_STATUS_TRANSITION}.
\end{itemize}

\subsection*{A.3 \code{UAT-OUM-03}: Forbidden Roles \& Security Boundaries}
\textbf{Purpose:} Verify backend RBAC enforcement and tenant data isolation.\\[1pt]
\textbf{Security Checkpoints:}
\begin{enumerate}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
  \item \textbf{Unauthenticated Request:} Calling \code{GET /api/requests/my} without cookie $\to$ HTTP 401 \code{AUTH_REQUIRED}.
  \item \textbf{Requester Review Attempt:} User with \code{REQUESTER} role posts to \code{/api/requests/:id/decision} $\to$ HTTP 403 \code{ACCESS_DENIED}.
  \item \textbf{Self-Approval Guard:} Reviewer creates request and attempts to approve own request $\to$ HTTP 403 \code{SELF_APPROVAL_NOT_ALLOWED}.
  \item \textbf{Unassigned Reviewer Denial:} Reviewer B attempts to decide request assigned to Reviewer A $\to$ HTTP 403 \code{ACCESS_DENIED}.
  \item \textbf{Cross-Requester Read Isolation:} Requester B requests \code{GET /api/requests/:id} of Requester A $\to$ HTTP 403 \code{ACCESS_DENIED}.
\end{enumerate}

\subsection*{A.4 \code{UAT-OUM-04}: State Persistence \& Immutability across Deployment}
\textbf{Purpose:} Guarantee data integrity across Cloud Run container redeployments and verify zero data mutation surfaces exist.\\[1pt]
\textbf{Procedure \& Audit:}
\begin{enumerate}[leftmargin=*,itemsep=1.5pt,topsep=1pt,parsep=0pt]
  \item Submit persistence marker request \code{REQ-UAT-PERSIST} on deployed revision \code{sd3cdaba1-00008-2pg}.
  \item Trigger deployment of revision \code{sd3cdaba1-00009-gxv} via Cloud Run console/CLI.
  \item Query \code{GET /api/requests/:id} on new revision $\to$ Request record, status, timestamps, and decision logs persist identically.
  \item Verify zero \code{DELETE} or \code{PUT/PATCH} endpoints are exposed on request entities (audit trail is strictly append-only).
\end{enumerate}
